💭 فكّر:

الساعات العادية تحتاج ضبطاً يدوياً — وإذا نفدت البطارية ترجع للصفر. كيف يمكن لجهاز أن يعرف الوقت الصحيح دائماً؟

🔮

توقّع أولاً

كيف يمكن لـ ESP32 أن يعرف الوقت الصحيح — بدون RTC، وبدون أن تضبطه أنت؟

✅ الإجابة:

عبر NTP = Network Time Protocol. ESP32 يتصل بالإنترنت، يسأل خادم وقت عالمي: "كم الساعة الآن؟" — فيرد عليه بالوقت بدقة أجزاء من الثانية.

الميزة الرائعة: نضبط المنطقة الزمنية (Time Zone) مرة واحدة في الكود، والوقت يُضبط تلقائياً إلى الأبد — حتى بعد انقطاع الكهرباء!

🔬 كيف يعمل NTP؟

03 / 11

💡 الفكرة في جملة

بدلاً من شراء شريحة RTC (تحتاج ضبط + بطارية)، نستخدم الإنترنت نفسه كمصدر وقت. ESP32 يسأل خادم NTP — والخادم يرد بالوقت الدقيق.

✅ مميزات:

• مجاني تماماً
• دقة متناهية (ms)
• لا يحتاج ضبطاً يدوياً
• يعمل حتى بعد انقطاع الكهرباء

2
يتصل بخادم NTP
pool.ntp.org — خوادم عالمية موزعة
3
يستقبل الوقت الدقيق
Time Stamp (Unix Time)
4
يضيف فرق المنطقة الزمنية
مصر = UTC +2
5
يعرض على OLED
صيغة: HH:MM:SS
🎯 لماذا "pool.ntp.org" وليس خادماً واحداً؟ لأن pool.ntp.org عبارة عن آلاف الخوادم موزعة عالمياً. ESP32 يختار أقرب خادم تلقائياً — أسرع وأكثر موثوقية.

🛡️ 3 قواعد أمنية

05 / 11
⚡ تحذير 1 — كلمة سر الواي فاي:

لا تضع كلمة السر مباشرة في الكود الذي تشاركه مع الآخرين. استخدم ملفاً منفصلاً secrets.h — يُشارَك مع الكود ملفٌ بديلٌ بدون كلمة السر الحقيقية.

⚠️ تحذير 2 — الشبكة المفتوحة:

لا تتصل بشبكات WiFi مفتوحة (بدون كلمة سر) من ESP32 — قد تكون شبكة خبيثة تعترض بياناتك.

⚠️ تحذير 3 — التحقق من الوقت:

في المشاريع الحرجة (مثل تسجيل الأحداث)، تحقق من نجاح المزامنة قبل الاعتماد على الوقت — وإلا سجّلت وقتاً خاطئاً.

💡 نصيحة احترافية: ضع شروط WiFi في ملف منفصل اسمه secrets.h: 
// secrets.h — لا ترفعه على GitHub!
#define WIFI_SSID  "MyHomeWiFi"
#define WIFI_PASS  "MySecret123"

const char* SSID     = "YourWiFiName";
const char* PASSWORD = "YourPassword";

void setup() {
  Serial.begin(115200);
  Serial.print("Connecting to WiFi");
  WiFi.begin(SSID, PASSWORD);

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println("\nConnected!");
  Serial.print("IP: ");
  Serial.println(WiFi.localIP());
}

void loop() {
  // لا شيء — هذا اختبار فقط
}

#include <WiFi.h>
#include <time.h>
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

#define SCREEN_W 128
#define SCREEN_H 64
#define OLED_ADDR 0x3C

const char* SSID     = "YourWiFiName";
const char* PASSWORD = "YourPassword";

// مصر: UTC+2 → "EET-2"
const char* TZ_INFO = "EET-2";
const char* NTP_SERVER = "pool.ntp.org";

Adafruit_SSD1306 display(SCREEN_W, SCREEN_H, &Wire, -1);

void connectWiFi() {
  Serial.print("Connecting WiFi");
  WiFi.begin(SSID, PASSWORD);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi OK");
}

void setup() {
  Serial.begin(115200);
  Wire.begin(21, 22);
  display.begin(SSD1306_SWITCHCAPVCC, OLED_ADDR);
  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);
  display.setTextSize(1);
  display.setCursor(0, 0);
  display.println("Connecting...");
  display.display();

  connectWiFi();

  // ضبط المنطقة الزمنية + خادم NTP
  configTime(0, 0, NTP_SERVER);
  setenv("TZ", TZ_INFO, 1);
  tzset();

  // انتظر أول مزامنة
  struct tm timeinfo;
  while (!getLocalTime(&timeinfo)) {
    Serial.println("Waiting for NTP...");
    delay(1000);
  }
  Serial.println("Time synced!");
}

void loop() {
  struct tm t;
  if (!getLocalTime(&t)) return;

  char timeStr[16];
  char dateStr[16];
  strftime(timeStr, sizeof(timeStr), "%H:%M:%S", &t);
  strftime(dateStr, sizeof(dateStr), "%d/%m/%Y", &t);

  display.clearDisplay();

  // العنوان
  display.setTextSize(1);
  display.setCursor(0, 0);
  display.println("ESP32 Smart Clock");

  // الوقت (كبير)
  display.setTextSize(3);
  display.setCursor(0, 18);
  display.println(timeStr);

  // التاريخ
  display.setTextSize(1);
  display.setCursor(0, 52);
  display.println(dateStr);

  display.display();
  delay(1000);
}
🎯 3 أفكار مفتاحية:

configTime() — يبدأ المزامنة مع خادم NTP.
setenv("TZ", ...) — يحدد المنطقة الزمنية.
getLocalTime(&t) — يقرأ الوقت المحلي في هيكل tm.
